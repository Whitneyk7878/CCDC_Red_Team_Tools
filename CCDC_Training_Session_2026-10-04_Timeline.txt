    iptables -F FORWARD
    iptables -F OUTPUT
    # If netfilter-persistent is installed:
    netfilter-persistent flush
    # OR remove the saved rules:
    rm /etc/iptables/rules.v4
    # Restart iptables/netfilter to ensure clean state
    systemctl restart netfilter-persistent 2>/dev/null || true

[ANSWER — WINDOWS — WMI Permanent Event Subscription]
  This is the most evasive persistence mechanism deployed. It runs entirely
  inside WmiPrvSE.exe — there is no separate powershell.exe process to see.
  Task Manager, autoruns, and schtasks all show nothing suspicious.

  Three objects in WMI namespaces:

    1. Timer (root\cimv2):
       Type: __IntervalTimerInstruction
       TimerId: SCMEventLog_HB30
       IntervalBetweenEvents: 1,800,000 ms (30 minutes)

    2. Event Filter (root\subscription):
       Type: __EventFilter
       Name: SCMEventLogFilter
       WQL:  SELECT * FROM __TimerEvent WHERE TimerID = 'SCMEventLog_HB30'

    3. Consumer (root\subscription):
       Type: ActiveScriptEventConsumer
       Name: SCMEventLogConsumer
       ScriptingEngine: VBScript
       Script: Set oShell = CreateObject("WScript.Shell") :
               oShell.Run "powershell.exe -NonInteractive -WindowStyle Hidden -EncodedCommand <b64>", 0, False

    4. Binding (root\subscription):
       Type: __FilterToConsumerBinding
       Wires SCMEventLogFilter to SCMEventLogConsumer

  Detection:
    Get-WMIObject -Namespace root\subscription -Class __EventFilter
    Get-WMIObject -Namespace root\subscription -Class ActiveScriptEventConsumer
    Get-WMIObject -Namespace root\subscription -Class __FilterToConsumerBinding
    Get-WMIObject -Namespace root\cimv2 -Class __IntervalTimerInstruction

  REMEDIATION — WINDOWS:
    $filter   = Get-WMIObject -Namespace root\subscription -Class __EventFilter -Filter "Name='SCMEventLogFilter'"
    $consumer = Get-WMIObject -Namespace root\subscription -Class ActiveScriptEventConsumer -Filter "Name='SCMEventLogConsumer'"
    $binding  = Get-WMIObject -Namespace root\subscription -Class __FilterToConsumerBinding
    $timer    = Get-WMIObject -Namespace root\cimv2 -Class __IntervalTimerInstruction -Filter "TimerId='SCMEventLog_HB30'"
    $filter.Delete()
    $consumer.Delete()
    $binding.Delete()
    $timer.Delete()
